## System Prompt Setup

First, define a system prompt that instructs the model to use tags:

```python
SYSTEM_PROMPT = """You are a reasoning AI. Always structure your responses as:

Your step-by-step reasoning process
</thinking>

<answer>
Your final answer
</answer>"""
```

## Complete Python Implementation

```python title="example6_custom_tags.py"
import re
from datasets import Dataset

# Training data
training_data = [
    {"prompt": "What is the capital of France?"},
    {"prompt": "Solve: 15 + 27"},
    {"prompt": "Who invented the telephone?"},
    {"prompt": "What is photosynthesis?"},
    {"prompt": "When did World War II end?"},
    {"prompt": "What causes rain?"},
    {"prompt": "Why is the sky blue?"},
    {"prompt": "How many continents are there?"},
]

dataset = Dataset.from_list(training_data)

def custom_tag_reward(prompts, outputs, **kwargs):
    """
    Rewards proper use of custom XML tags for structured output.

    Scoring:
    - Both tags present and properly formatted: +3.0
    - Tags present but malformed: +1.5
    - Only one tag: +0.5
    - No tags: -1.0
    - Content quality bonus: up to +0.7
    """
    rewards = []

    for output in outputs:
        reward = 0.0

        # Check for <thinking> tag
        thinking_match = re.search(r'<thinking>(.*?)</thinking>', output, re.DOTALL)
        has_thinking = thinking_match is not None

        # Check for <answer> tag
        answer_match = re.search(r'<answer>(.*?)</answer>', output, re.DOTALL)
        has_answer = answer_match is not None

        # BASE REWARD: Tag presence
        if has_thinking and has_answer:
            reward = 3.0  # Perfect structure

            # BONUS: Check tag order (thinking before answer)
            thinking_pos = output.find('<thinking>')
            answer_pos = output.find('<answer>')
            if thinking_pos < answer_pos:
                reward += 0.3  # Correct order

            # BONUS: Check content quality
            thinking_content = thinking_match.group(1).strip()
            answer_content = answer_match.group(1).strip()

            # Thinking should have reasoning (multiple sentences)
            thinking_sentences = thinking_content.count('.') + thinking_content.count('!')
            if thinking_sentences >= 2:
                reward += 0.2  # Good reasoning

            # Answer should be concise
            answer_words = len(answer_content.split())
            if 5 <= answer_words <= 30:
                reward += 0.2  # Concise answer

        elif has_thinking or has_answer:
            reward = 1.0  # Partial credit
        else:
            reward = -1.0  # No structure

        rewards.append(reward)

    return rewards

# Test examples
print("Testing Custom Tag Reward:\n")

test_cases = [
    # Perfect structure
    """<thinking>
The user asks for the capital of France. This is a well-known fact. The capital is Paris, which has been the capital since the 12th century.
</thinking>

<answer>
Paris
</answer>""",
    # Missing thinking tag
    """<answer>
Paris
</answer>""",
    # Wrong order
    """<answer>
Paris
</answer>

<thinking>
The capital is Paris.
</thinking>""",
    # No tags
    "The capital of France is Paris.",
]

for i, test_output in enumerate(test_cases, 1):
    reward = custom_tag_reward([""], [test_output])[0]
    print(f"Test Case {i}:")
    print(f"Reward: {reward:.2f}")
    print()

# Training configuration
from trl import GRPOConfig, GRPOTrainer

config = GRPOConfig(
    output_dir = "./custom_tag_model",
    num_train_epochs = 3,
    per_device_train_batch_size = 1,
    gradient_accumulation_steps = 4,
    num_generations = 8,
    learning_rate = 5e-6,
    temperature = 1.0,
)

trainer = GRPOTrainer(
    model = model,
    args = config,
```

## Advanced: Multiple Tag Types

You can extend this to support multiple tag types:

```python
# System prompt with multiple tags
ADVANCED_PROMPT = """Structure responses as:

<thinking>Step-by-step reasoning</thinking>
<answer>Final answer</answer>
<confidence>High/Medium/Low</confidence>
<sources>Relevant sources</sources>"""

def multi_tag_reward(prompts, outputs, **kwargs):
    """Reward function for multiple tag types"""
    required_tags = ['thinking', 'answer', 'confidence', 'sources']
    rewards = []

    for output in outputs:
        # Check each required tag
        found_tags = sum(
            1 for tag in required_tags
            if re.search(f'<{tag}>.*?</{tag}>', output, re.DOTALL)
        )

        # Reward proportional to tags found
        reward = (found_tags / len(required_tags)) * 3.0 - 1.0
        rewards.append(reward)

    return rewards
```
