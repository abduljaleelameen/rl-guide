            # Check if validation reward is improving

            # Log to wandb

    ### Canary Deployment with A/B Testing

    ```python title="deployment_pipeline.py"
    import requests
    from datetime import datetime
    import json

    class ModelDeployment:
        def __init__(self, baseline_url, candidate_url):
            self.baseline_url = baseline_url
            self.candidate_url = candidate_url
            self.traffic_split = 0.05  # Start with 5%
            self.metrics = {
                'baseline': [],
                'candidate': [],
            }

        def route_request(self, prompt):
            """Route requests with traffic splitting"""
            import random

            # Traffic splitting
            use_candidate = random.random() < self.traffic_split

            if use_candidate:
                response = requests.post(
                    f"{self.candidate_url}/generate",
                    json={"prompt": prompt}
                )
                model_version = "candidate"
            else:
                response = requests.post(
                    f"{self.baseline_url}/generate",
                    json={"prompt": prompt}
                )
                model_version = "baseline"

            # Log the interaction
            self.log_interaction(
                prompt=prompt,
                response=response.json(),
                model=model_version,
                timestamp=datetime.now()
            )

            return response.json()

        def log_interaction(self, prompt, response, model, timestamp):
            """Log every interaction for analysis"""
            log_entry = {
                'prompt': prompt,
                'response': response,
                'model': model,
                'timestamp': timestamp.isoformat(),
                'latency': response.get('latency_ms'),
            }

            with open(f'logs/{model}_interactions.jsonl', 'a') as f:
                f.write(json.dumps(log_entry) + '\n')

        def calculate_statistical_significance(self):
            """Check if candidate is significantly better"""
            from scipy import stats

            baseline_rewards = self.metrics['baseline']
            candidate_rewards = self.metrics['candidate']

            if len(baseline_rewards) < 100 or len(candidate_rewards) < 100:
                return False, "Insufficient data"

            # T-test
            t_stat, p_value = stats.ttest_ind(candidate_rewards, baseline_rewards)

            # Candidate must be significantly better (p < 0.05)
            is_better = (
                p_value < 0.05 and
                np.mean(candidate_rewards) > np.mean(baseline_rewards)
            )

            return is_better, p_value

        def gradual_rollout(self):
            """Gradually increase traffic to candidate"""
            # 5% → 25% → 50% → 100%
            rollout_stages = [0.05, 0.25, 0.50, 1.00]

            for stage in rollout_stages:
                print(f"Setting traffic to {stage * 100}% candidate")
                self.traffic_split = stage

                # Run for N requests
                for _ in range(1000):
                    # Simulate traffic
                    self.route_request("test prompt")

                # Check metrics
                is_better, p_value = self.calculate_statistical_significance()

                if not is_better:
                    print(f"⚠️ Rollback! Candidate not better (p={p_value:.3f})")
                    self.rollback()
                    return False

                print(f"✅ Stage {stage} passed (p={p_value:.3f})")

            print("🚀 Full rollout complete!")
            return True

        def rollback(self):
            """Immediate rollback to baseline"""
            print("🔄 Rolling back to baseline model...")
            self.traffic_split = 0.0

            # Alert team
            self.send_alert(
                severity="high",
                message="Candidate model rolled back due to poor performance"
            )

    # Example usage
    deployment = ModelDeployment(
        baseline_url="http://baseline-server:8000",
        candidate_url="http://candidate-server:8000"
    )

    # Start gradual rollout
    success = deployment.gradual_rollout()
    ```

    ### Real-Time Monitoring Dashboard

    ```python title="monitoring.py"
    import prometheus_client
    from prometheus_client import Counter, Histogram, Gauge
    import time

    # Define metrics
    request_count = Counter(
        'model_requests_total',
        'Total requests',
        ['model_version', 'status']
    )

    request_latency = Histogram(
        'model_latency_seconds',
        'Request latency',
        ['model_version']
    )

    reward_metric = Gauge(
        'model_reward_current',
        'Current reward score',
        ['model_version']
    )

    class ProductionMonitor:
        def __init__(self, alert_webhook):
            self.alert_webhook = alert_webhook
            self.reward_history = []
            self.alert_threshold = 0.8  # 80% of baseline

        def track_request(self, model_version, prompt, response, reward):
            """Track every request"""
            # Prometheus metrics
            request_count.labels(
                model_version=model_version,
                status="success"
            ).inc()

            request_latency.labels(
                model_version=model_version
            ).observe(response['latency_ms'] / 1000)

            reward_metric.labels(
                model_version=model_version
            ).set(reward)

            # Check for anomalies
            self.check_anomalies(model_version, reward)

        def check_anomalies(self, model_version, current_reward):
            """Detect degradation"""
            if len(self.reward_history) < 100:
                self.reward_history.append(current_reward)
                return

            # Rolling average
            avg_reward = np.mean(self.reward_history[-100:])

            # Alert if current reward drops significantly
            if current_reward < avg_reward * self.alert_threshold:
                self.send_alert(
                    severity="critical",
                    message=f"Reward dropped {((avg_reward - current_reward) / avg_reward) * 100:.1f}%",
                    model=model_version,
                    current_reward=current_reward,
                    baseline_reward=avg_reward
                )

            self.reward_history.append(current_reward)

        def send_alert(self, severity, message, **kwargs):
            """Send alert to Slack/PagerDuty"""
            alert_data = {
                'severity': severity,
                'message': message,
                'timestamp': datetime.now().isoformat(),
                **kwargs
            }

            # Send to webhook (Slack, PagerDuty, etc.)
            requests.post(self.alert_webhook, json=alert_data)

            print(f"🚨 ALERT [{severity}]: {message}")

    # Start Prometheus server
    prometheus_client.start_http_server(9090)
    print("Monitoring server started on :9090")
    ```

## Monitoring Metrics

### Critical Metrics

| Metric | Threshold | Action |
|--------|-----------|--------|
| Avg Reward | < 80% of baseline | Rollback |
| Latency p99 | > 2 seconds | Alert |
| Error Rate | > 1% | Investigate |
| Token/sec | < 500 | Scale up |

### Business Metrics

- User satisfaction (thumbs up/down)
- Task completion rate
- Average session length
- Retention rate

        self.baseline = baseline_model
        self.candidate = candidate_model

## Common Production Issues

### Issue: Model drift over time
**Solution:** Weekly retraining on fresh data

### Issue: Catastrophic failures in production
**Solution:** Extensive pre-deployment testing + canary rollout

### Issue: Unclear why model was rolled back
**Solution:** Comprehensive logging and metric tracking
