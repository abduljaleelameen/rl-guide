## Complete Python Implementation

```python title="example1_sentiment.py"
# Install sentiment analysis library
!pip install textblob

from textblob import TextBlob
from datasets import Dataset

# ========================================
# STEP 1: Create Training Data
# ========================================
# Create prompts that could have positive or negative responses
training_data = [
    {"prompt": "Tell me about your day."},
    {"prompt": "How do you feel about challenges?"},
    {"prompt": "What's your opinion on rainy weather?"},
    {"prompt": "Describe your approach to problems."},
    {"prompt": "How do you handle difficult situations?"},
    {"prompt": "What do you think about Mondays?"},
    {"prompt": "Tell me about working hard."},
    {"prompt": "How do you feel when things go wrong?"},
    {"prompt": "Describe a typical morning."},
    {"prompt": "What's your view on learning new things?"},
]

dataset = Dataset.from_list(training_data)

# ========================================
# STEP 2: Define Sentiment Reward Function
# ========================================
def sentiment_reward_function(prompts, outputs, **kwargs):
    """
    Reward function that encourages positive sentiment.

    How it works:
    1. Analyze sentiment of each output (-1 to +1)
    2. Scale sentiment to reward range (-3 to +3)
    3. Add bonus for very positive responses

    Args:
        prompts: List of input questions
        outputs: List of model responses

    Returns:
        rewards: List of reward scores
    """
    rewards = []

    for output in outputs:
        # Analyze sentiment using TextBlob
        # polarity ranges from -1.0 (negative) to +1.0 (positive)
        blob = TextBlob(output)
        sentiment_score = blob.sentiment.polarity

        # Base reward: scale sentiment to our reward range
        reward = sentiment_score * 3.0

        # Bonus for very positive responses (> 0.5)
        if sentiment_score > 0.5:
            reward += 0.5  # Extra reward for being super positive!

        # Penalty for negative responses (< -0.2)
        if sentiment_score < -0.2:
            reward -= 1.0  # Discourage negativity

        # Add small bonus if response is longer (more thoughtful)
        word_count = len(output.split())
        if word_count > 20 and word_count < 100:
            reward += 0.2  # Sweet spot: detailed but not rambling

        rewards.append(reward)

    return rewards

# ========================================
# STEP 3: Test the Reward Function
# ========================================
print("Testing reward function with sample outputs:\n")

test_outputs = [
    "I had an amazing day! Everything went perfectly and I'm so happy!",
    "It was okay. Nothing special.",
    "Everything went wrong. I'm frustrated and annoyed.",
]

test_rewards = sentiment_reward_function([], test_outputs)

for i, (output, reward) in enumerate(zip(test_outputs, test_rewards)):
    sentiment = TextBlob(output).sentiment.polarity
    print(f"Example {i+1}:")
    print(f"  Output: {output}")
    print(f"  Sentiment Score: {sentiment:.2f}")
    print(f"  Reward: {reward:.2f}")
    print()

# ========================================
# STEP 4: Train with GRPO
# ========================================
from trl import GRPOConfig, GRPOTrainer

config = GRPOConfig(
    output_dir = "./sentiment_model",
    num_train_epochs = 2,
    per_device_train_batch_size = 1,
    gradient_accumulation_steps = 4,
    num_generations = 8,
    learning_rate = 5e-6,
    temperature = 1.2,
)

trainer = GRPOTrainer(
    model = model,  # Assumes model loaded from previous section

    tokenizer = tokenizer,

trainer.train()
print("✅ Model now trained to be positive!")
```
